# Import all next-level subdirectories that have a CMakeLists.txt file

file(GLOB subdir_cmake_files RELATIVE ${CMAKE_CURRENT_LIST_DIR} ${CMAKE_CURRENT_LIST_DIR}/*/CMakeLists.txt)
foreach(cmake_file ${subdir_cmake_files})
    get_filename_component(dir ${cmake_file} DIRECTORY)
    add_subdirectory(${dir})
endforeach()
